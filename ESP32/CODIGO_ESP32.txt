#include <WiFi.h>
#include <WiFiClientSecure.h>
#include <HTTPClient.h>

// ===== EDITA ESTAS 3 LINEAS =====
const char* SSID   = "WIFI_NAME";
const char* PASS   = "123456789";
const char* DB_URL = "https://creias_que_iba_a_subir_mi_link?.firebaseio.com/datos.json";
// ================================



const float R0 = 1655.0;   // MQ-7: calibrado en la U


WiFiClientSecure client;

void setup() {
  Serial.begin(115200);
  pinMode(25, OUTPUT);  // relé bomba 1
  pinMode(26, OUTPUT);  // relé bomba 2
  pinMode(27, OUTPUT);  // buzzer
  digitalWrite(25, HIGH);  // bomba 1 apagada
  digitalWrite(26, HIGH);  // bomba 2 apagada

  Serial.println("Calentando MQ-7 (3 sec)...");
  delay(3000);  // esperar a que el sensor de gas se caliente


  // WiFi
  WiFi.begin(SSID, PASS);
  Serial.print("Conectando WiFi");
  while (WiFi.status() != WL_CONNECTED) { Serial.print("."); delay(300); }
  Serial.println(" OK");
  client.setInsecure();
}

void loop() {
  // Temperatura (LM35, calibrado)
  float mvT = analogReadMilliVolts(34);
  float temp = (mvT - 78.05) / 8.627;

  // Gas (MQ-7, calibrado)
  float v = analogReadMilliVolts(35) / 1000.0;
  float rs = 1500.0 * (5.0 - v) / v;
  float ratio = rs / R0;

  // Decisiones
  bool gas = ratio < 10;    // hay gas
  bool calor = temp > 45;   // temperatura alta
  

  if (gas) {
    digitalWrite(27, HIGH);  // buzzer encendido
  } else {
    digitalWrite(27, LOW);   // buzzer apagado
  }


  if (calor) {
    digitalWrite(25, LOW);   // bomba 1 encendida
    digitalWrite(26, LOW);   // bomba 2 encendida
  } else {
    digitalWrite(25, HIGH);  // bomba 1 apagada
    digitalWrite(26, HIGH);  // bomba 2 apagada
  }


  Serial.print("Temp: ");   Serial.print(temp);
  Serial.print("  Rs/R0: "); Serial.println(ratio);

  enviarFirebase(temp, ratio, gas, calor);
  
  delay(1000);
}








void enviarFirebase(float temp, float ratio, bool gas, bool calor) {
  if (WiFi.status() != WL_CONNECTED) return;

  String json = "{";
  json += "\"temp\":"    + String(temp, 2);
  json += ",\"ratio\":"  + String(ratio, 2);
  json += ",\"gas\":"    + String(gas   ? "true" : "false");
  json += ",\"calor\":"  + String(calor ? "true" : "false");
  json += "}";

  HTTPClient http;
  http.begin(client, DB_URL);
  http.addHeader("Content-Type", "application/json");
  int code = http.PUT(json);
  if (code != 200) { Serial.print("  [Firebase error] "); Serial.println(code); }
  http.end();
}
